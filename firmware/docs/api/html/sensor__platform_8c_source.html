83static void detect_motion(void)
84{
85 const uint8_t addresses[] = {0x68, 0x69};
86 for (size_t i = 0; i < 2; ++i) {
87 if (!s_address_present[addresses[i]]) continue;
88 i2c_master_dev_handle_t device = NULL;
89 if (add_device(addresses[i], &device) != ESP_OK) continue;
90 uint8_t id = 0;
91 if (read_register(device, 0x75, &id, 1) == ESP_OK) {
92 ESP_LOGI(TAG, "motion 0x%02X WHO_AM_I=0x%02X (%s)", addresses[i], id,
93 id == 0x68 ? "MPU-6050" : "unknown motion chip");
94 if (id == 0x68 && !s_mpu_present) {
95 s_mpu = device;
96 s_mpu_present = true;
97 if (write_register(0x6B, 0x00) != ESP_OK ||
98 write_register(0x1C, 0x00) != ESP_OK ||
99 write_register(0x1A, APP_MPU_DLPF_CFG) != ESP_OK ||
100 write_register(0x19, APP_MPU_SAMPLE_RATE_DIV) != ESP_OK) {
101 ESP_LOGE(TAG, "MPU-6050 configuration failed");
102 }
103 } else {
104 i2c_master_bus_rm_device(device);
105 }
106 } else {
107 i2c_master_bus_rm_device(device);
108 }
109 }
110 ESP_LOGI(TAG, "MPU-6050 detected: %s", s_mpu_present ? "yes" : "no");

138static void read_bmp180_calibration(void)
139{
140 uint8_t raw[22];
141 if (!s_pressure_present || read_register(s_pressure, 0xAA, raw, sizeof(raw)) != ESP_OK) return;
142 s_ac1 = (int16_t)((raw[0] << 8) | raw[1]); s_ac2 = (int16_t)((raw[2] << 8) | raw[3]);
143 s_ac3 = (int16_t)((raw[4] << 8) | raw[5]); s_ac4 = (uint16_t)((raw[6] << 8) | raw[7]);
144 s_ac5 = (uint16_t)((raw[8] << 8) | raw[9]); s_ac6 = (uint16_t)((raw[10] << 8) | raw[11]);
145 s_b1 = (int16_t)((raw[12] << 8) | raw[13]); s_b2 = (int16_t)((raw[14] << 8) | raw[15]);
146 s_mb = (int16_t)((raw[16] << 8) | raw[17]); s_mc = (int16_t)((raw[18] << 8) | raw[19]);
147 s_md = (int16_t)((raw[20] << 8) | raw[21]);

150static void update_temperature(void)
151{
152 if (!s_pressure_present) return;
153 uint8_t command[] = {0xF4, 0x2E};
154 if (i2c_master_transmit(s_pressure, command, sizeof(command), APP_I2C_TIMEOUT_MS) != ESP_OK) return;
155 vTaskDelay(pdMS_TO_TICKS(5));
156 uint8_t raw[2];
157 if (read_register(s_pressure, 0xF6, raw, sizeof(raw)) != ESP_OK) return;
158 const int32_t ut = (raw[0] << 8) | raw[1];
159 const int32_t x1 = ((ut - s_ac6) * s_ac5) >> 15;
160 const int32_t x2 = (s_mc << 11) / (x1 + s_md);
161 const int32_t b5 = x1 + x2;
162 s_temperature_c = ((float)(b5 + 8) / 16.0f) / 10.0f;
163 s_temperature_valid = isfinite(s_temperature_c);
164}

186esp_err_t vibration_source_read(vibration_sample_t *sample)
187{
188 if (!sample) return ESP_ERR_INVALID_ARG;
189#if APP_SENSOR_SIMULATION
190 const uint64_t now_ms = (uint64_t)(esp_timer_get_time() / 1000ULL);
191 if (s_simulation_start_ms == 0) s_simulation_start_ms = now_ms;
192 const uint32_t cycle_ms = APP_SENSOR_SIM_OFF_MS + APP_SENSOR_SIM_STARTING_MS +
193 APP_SENSOR_SIM_RUNNING_MS + APP_SENSOR_SIM_STOPPING_MS;
194 const uint32_t phase_ms = (uint32_t)((now_ms - s_simulation_start_ms) % cycle_ms);
195 const uint32_t start = APP_SENSOR_SIM_OFF_MS;
196 const uint32_t run = start + APP_SENSOR_SIM_STARTING_MS;
197 const uint32_t stop = run + APP_SENSOR_SIM_RUNNING_MS;
198 float value = 0.01f;
199 if (phase_ms >= start && phase_ms < run) {
200 value = 0.01f + 0.79f * (float)(phase_ms - start) / (float)APP_SENSOR_SIM_STARTING_MS;
201 } else if (phase_ms >= run && phase_ms < stop) {
202 value = 0.05f;
203 } else if (phase_ms >= stop) {
204 value = 0.40f * (1.0f - (float)(phase_ms - stop) / (float)APP_SENSOR_SIM_STOPPING_MS);
205 }
206 sample->x_g = value; sample->y_g = 0.0f; sample->z_g = 0.0f; sample->magnitude_g = value;
207 return ESP_OK;
208#else
209 if (!s_mpu_present) return ESP_ERR_INVALID_STATE;
210 uint8_t raw[6];
211 esp_err_t err = read_register(s_mpu, 0x3B, raw, sizeof(raw));
212 if (err != ESP_OK) return err;
213 const int16_t x = (int16_t)((raw[0] << 8) | raw[1]);
214 const int16_t y = (int16_t)((raw[2] << 8) | raw[3]);
215 const int16_t z = (int16_t)((raw[4] << 8) | raw[5]);
216 const float ax = (float)x * APP_MPU_ACCEL_SCALE_G / 32768.0f;
217 const float ay = (float)y * APP_MPU_ACCEL_SCALE_G / 32768.0f;
218 const float az = (float)z * APP_MPU_ACCEL_SCALE_G / 32768.0f;
219 const float gravity_alpha = 0.01f;
220 s_gravity_x += gravity_alpha * (ax - s_gravity_x);
221 s_gravity_y += gravity_alpha * (ay - s_gravity_y);
222 s_gravity_z += gravity_alpha * (az - s_gravity_z);
223 sample->x_g = ax - s_gravity_x; sample->y_g = ay - s_gravity_y; sample->z_g = az - s_gravity_z;
224 sample->magnitude_g = sqrtf(sample->x_g * sample->x_g + sample->y_g * sample->y_g + sample->z_g * sample->z_g);
225 return ESP_OK;
226#endif
227}

229static void sensor_task(void *arg)
230{
231 (void)arg;
232 float sum_sq = 0.0f, peak = 0.0f;
233 uint32_t count = 0, errors = 0, last_temp_ms = 0;
234 operating_state_init();
235 anomaly_detector_init();
236#if APP_SENSOR_LEARNING_ENABLE
237 baseline_learner_init(APP_SENSOR_BASELINE_REQUIRED_CYCLES);
238#endif
239 for (;;) {
240 const uint64_t now_us = esp_timer_get_time();
241 vibration_sample_t sample;
242 if (vibration_source_read(&sample) == ESP_OK && isfinite(sample.magnitude_g)) {
243 if (!s_filter_initialized) { s_filtered_magnitude = sample.magnitude_g; s_filter_initialized = true; }
244 else s_filtered_magnitude = APP_SENSOR_EMA_ALPHA * sample.magnitude_g + (1.0f - APP_SENSOR_EMA_ALPHA) * s_filtered_magnitude;
245 sum_sq += s_filtered_magnitude * s_filtered_magnitude;
246 if (s_filtered_magnitude > peak) peak = s_filtered_magnitude;
247 ++count;
248 } else ++errors;
249 const uint32_t now_ms = (uint32_t)(now_us / 1000ULL);
250 if (now_ms - last_temp_ms >= APP_SENSOR_TEMP_PERIOD_MS) { update_temperature(); last_temp_ms = now_ms; }
251 if (now_ms % APP_SENSOR_WINDOW_MS < APP_SENSOR_RAW_PERIOD_MS && count > 0) {
252 const float rms = sqrtf(sum_sq / (float)count);
253 const sentinel_state_t state = operating_state_update(rms, now_ms);
254 const anomaly_level_t anomaly = state == SENTINEL_STATE_RUNNING ?
255 anomaly_detector_update(rms, now_ms, baseline_learner_is_valid()) : ANOMALY_NONE;
256 if (anomaly == ANOMALY_CRITICAL || state == SENTINEL_STATE_FAULT) led_platform_set_status(LED_STATUS_CRITICAL);
257 else if (anomaly == ANOMALY_WARNING) led_platform_set_status(LED_STATUS_WARNING);
258 else if (state == SENTINEL_STATE_FAULT) led_platform_set_status(LED_STATUS_CRITICAL);
259 else if (baseline_learner_is_learning()) led_platform_set_status(LED_STATUS_BASELINE);
260 else if (state == SENTINEL_STATE_RUNNING) led_platform_set_status(LED_STATUS_RUNNING);
261 else if (state == SENTINEL_STATE_OFF) led_platform_set_status(LED_STATUS_OFF);
262 else led_platform_set_status(LED_STATUS_BOOT);
263#if APP_SENSOR_LEARNING_ENABLE
264 baseline_learner_observe(state, rms);
265#endif
266 telemetry_snapshot_t snapshot = {
267 .uptime_ms = now_ms,
268 .vibration_rms_g = rms,
269 .vibration_peak_g = peak,
270 .temperature_c = s_temperature_c,
271 .temperature_valid = s_temperature_valid,
272 .samples = count,
273 .errors = errors,
274 .phase = state,
275 .anomaly = anomaly
276 };
277 telemetry_publish(&snapshot);
278 ESP_LOGI(TAG, "uptime=%lu phase=%s rms=%.3f g peak=%.3f g temp=%s%.2f C samples=%lu errors=%lu anomaly=%s",
279 (unsigned long)now_ms, operating_state_name(state), rms, peak,
280 s_temperature_valid ? "" : "invalid ", s_temperature_c,
281 (unsigned long)count, (unsigned long)errors,
282 anomaly_detector_name(anomaly));
283 sum_sq = 0.0f; peak = 0.0f; count = 0; errors = 0;
284 }
285 ulTaskNotifyTake(pdTRUE, portMAX_DELAY);
286 }
