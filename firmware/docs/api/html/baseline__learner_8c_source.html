36static void calculate_thresholds(void)
37{
38 const baseline_phase_stats_t *running = &s_stats[SENTINEL_STATE_RUNNING];
39 const float variance = running->windows > 1 ? running->m2_g / (running->windows - 1U) : 0.0f;
40 const float sigma = sqrtf(fmaxf(0.0f, variance));
41 s_warning_g = fmaxf(0.08f, running->mean_g + 3.0f * sigma);
42 s_critical_g = fmaxf(0.30f, running->mean_g + 5.0f * sigma);
43 if (s_critical_g <= s_warning_g) s_critical_g = s_warning_g + 0.10f;
44}

59void baseline_learner_observe(sentinel_state_t state, float rms_g)
60{
61 if (!s_learning || s_valid || !isfinite(rms_g) || state > SENTINEL_STATE_FAULT) return;
62 baseline_phase_stats_t *stats = &s_stats[state];
63 ++stats->windows;
64 const float delta = rms_g - stats->mean_g;
65 stats->mean_g += delta / (float)stats->windows;
66 stats->m2_g += delta * (rms_g - stats->mean_g);
67 if (rms_g < stats->min_g) stats->min_g = rms_g;
68 if (rms_g > stats->max_g) stats->max_g = rms_g;
69}
